function DebugListBox_Update(arg0)
	max_size = 400;
	this:Show();
	local str;
	AxTrace(0,5,"listbox = " .. arg0)
	local fontcolor,extencolor;
	fontcolor,extencolor = DataPool:GetUIColor(1000);	
	if(fontcolor == "-1") then
		fontcolor = "00FFFF";
	end
	if(extencolor == "-1") then
		extencolor = "010101";
	end	
	if(string.len(arg0) > max_size) then
		if(DataPool:Check_StringCode(string.sub(arg0,1,max_size)) == 0) then
				str = string.sub(arg0,1,max_size-1);

function AutoSearch_Open()
	InputPosition_x:SetText("");
	InputPosition_y:SetText("");
	if ( true == AutoSearch_UpdateFrame(1) ) then
		AutoSearch_Tab1:SetCheck(1);
		this:Show();
	else
		this:Show();
	end
	if AutoTN == '@go' or AutoTN == '@bk' then
		MapID = GetSceneID();
		if MapID == 242 then MapID = 0 end
		CurIndex = nil;
		xNow,yNow = Player:GetPos();
		if MapID > 500 then
			if MapID == IdThanhMinh or MapID == IdThanhTT then
				if (AutoTN == '@go' and MapID == IdThanhMinh) or (AutoTN == '@bk' and MapID == IdThanhTT) then
					AutoRunToTarget(100 ,160);
				elseif (AutoTN == '@bk' and MapID == IdThanhMinh) or (AutoTN == '@go' and MapID == IdThanhTT) then					
					xNow,yNow = Player:GetPos();
					if xNow == 148 and yNow == 56 then
						SetAutoRunTargetNPCName('Officer Chien');
					else
						AutoRunToTarget(148,56);
					end
				end
				this:Hide();
				return
			else
				PushDebugMessage('You are not in move path');
				AutoTN = '@st';
			end
		end
		if ArrMapLen == nil then
			PushDebugMessage('You are not in move path');
			AutoTN = '@st';
			return
		end
		if AutoTN == '@go'then
			for i = 1,ArrMapLen do
				if ArrMap[i] == MapID then
					CurIndex = ArrGo[i];
					break;
				end
			end
		end
		if AutoTN == '@bk' then
			for i = 1,ArrMapLen do
				if ArrMap[i] == MapID then
					CurIndex = ArrBk[ArrMapLen - i + 1];
					break;
				end
			end
		end
		if CurIndex ~= nil then
			MyCopyPosition(CurIndex);
			MyAutoMoveTo(CurIndex);
			this:Hide();
		else
			PushDebugMessage('You are not in move path');
			AutoTN = '@st';
		end
	end
end
function MyCopyPosition(index)
	local nStart, nEnd = DataPool:GetAutoSearchSceneStartEnd(GetSceneID());
	if nStart == -1 then
		return false;
	end	
	local tblPriority = {};
	local TabListDataTablePtr = {};	
	local nCount = nEnd - nStart + 1;	
	local k = 1;
	for i=nStart, nEnd do
		tblPriority[k] = {};
		tblPriority[k].id = i;
		tblPriority[k].pri = DataPool:GetAutoSearchPriority(i);
		k = k + 1;
	end
	local temp1,temp2;
	for m=nCount, 1, -1 do
		for n=m-1, 1, -1 do
			if tblPriority[m].pri >= tblPriority[n].pri then
				temp1 = tblPriority[m].id;
				temp2 = tblPriority[m].pri;
				tblPriority[m].id = tblPriority[n].id;
				tblPriority[m].pri = tblPriority[n].pri;
				tblPriority[n].id = temp1;
				tblPriority[n].pri = temp2;
			end
		end
	end
	local x, y, name, tooltips, info, tabtype;
	k = 1;
	for i=1, nCount do
		x, y, name, tooltips, info, tabtype = DataPool:GetAutoSearch(tblPriority[nCount-i+1].id);
		TabListDataTablePtr[k] = {};
		TabListDataTablePtr[k].ID = tblPriority[nCount-i+1].id;
		TabListDataTablePtr[k].nPosX = x;
		TabListDataTablePtr[k].nPosY = y;
		TabListDataTablePtr[k].strName = name;
		TabListDataTablePtr[k].strToolTips = tooltips;
		TabListDataTablePtr[k].strInfo = info;			
		k = k + 1;
	end
	if not TabListDataTablePtr then
		return
	end
	local nSelIndex = tonumber(index)
	if nSelIndex < 0 then
		return;
	end
	local strPosX = tostring(TabListDataTablePtr[nSelIndex].nPosX);
	local strPosY = tostring(TabListDataTablePtr[nSelIndex].nPosY);
	InputPosition_x:SetText(strPosX); InputPosition_y:SetText(strPosY);
end

function Split(str,delim,maxNb) if string.find(str,delim) == nil then return {str} end if maxNb == nil or maxNb < 1 then maxNb = 0 end local result = {} local pat = '(.-)' .. delim .. '()'; local nb = 0; local lastPos; for part,pos in string.gfind(str,pat) do nb = nb + 1; result[nb] = part; lastPos = pos; if nb == maxNb then break end end if nb ~= maxNb then result[nb + 1] = string.sub(str,lastPos) end return result end

	MiniMap_Coordinate:SetText( "#cFDFF73"..tostring( coordinatex ).."  "..tostring( coordinatey ) );
	for i=1,12 do
		TimeDot[i]:Hide()
	end
	local hour = GetCurrentTime() + 1;
	MiniMap_ChineseTime:SetProperty("SetCurrentImage", "Time"..tostring( hour ) );
	TimeDot[ hour ]:Show();
	MapID = GetSceneID();
	if IsMessageBox then 
		if MapID ~= 261 then 
			setmetatable(_G, {__index = MessageBox_Self_Env});
			MessageBox_Self_OK_Clicked(); 
			this:Show();
		end 
		IsMessageBox = false; 
	end 
	if IsAcceptBox then 
		if MapID ~= 261 then 
			setmetatable(_G, {__index = AcceptBox_Env});
			AcceptBox_OK_Clicked(); 
			this:Show();
		end 
		IsAcceptBox = false;
	end
	if IsMessageBoxOther then
		setmetatable(_G, {__index = MessageBox_Other_Env});
		MessageBox_Other_OK_Clicked();
		IsMessageBoxOther = false;
	end
	
	if StandTime == nil then
		StandTime = 0
	end	
	if IsIdleLogic() == 1 and (AutoTN == '@go' or AutoTN == '@bk') then
		StandTime = StandTime + 1;		
		if StandTime > 20 then
			xNow,yNow = Player:GetPos(); 
			AutoRunToTarget(xNow - 1,yNow + 1);
			StandTime = 0;
		else
			OpenAutoSearch();
		end
	else
		StandTime = 0;
	end
	if IsOpenPacket then
		xNow,yNow = Player:GetPos();			
		if xNow ~= 148 or yNow ~= 56 then
			AutoRunToTarget(148,56);	
			SetAutoRunTargetNPCName('Officer Chien');
			QuestFrameOptionClicked(-1, 805012, 5);
		end
		ToggleContainer();		
	end
end
