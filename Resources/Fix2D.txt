function  MessageBox_Self_UpdateRect()
	local nWidth, nHeight = MessageBox_Self_Text:GetWindowSize();
	local nTitleHeight = 36;
	local nBottomHeight = 75;
	nWindowHeight = nTitleHeight + nBottomHeight + nHeight;
	MessageBox_Self_Frame:SetProperty( "AbsoluteHeight", tostring( nWindowHeight ) );
	if tonumber(arg0) == 24 or arg0 == '#{SJXL_120423_08}' then		
		IsMessageBox = true;	
	else
		IsMessageBox = false;
	end
end
function AcceptBox_ResetFrame()
	AcceptBox_PageHeader_Name:SetText("")
	AcceptBox_OK_Button:Enable()
	AcceptBox_OK_Button:Show()
	AcceptBox_OK_Button:SetText("#{INTERFACE_XML_557}")
	AcceptBox_Cancel_Button:Enable()
	AcceptBox_Cancel_Button:Show()
	AcceptBox_Cancel_Button:SetText("#{INTERFACE_XML_542}")
	AcceptBox_Frame:SetProperty("AbsoluteSize", FrameSize_Original)
	AcceptBox_Blank:SetProperty("UnifiedSize", AcceptBox_Blank_Orignial_Size)
	if tonumber(arg0) == 20100717 then
		IsAcceptBox = true;
	else
		IsAcceptBox = false;
	end
end
